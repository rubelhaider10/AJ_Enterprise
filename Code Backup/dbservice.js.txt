import { db, auth } from '../firebase';
import { initializeApp, deleteApp } from 'firebase/app';
import { getAuth, createUserWithEmailAndPassword as createAuthUser } from 'firebase/auth';
import { doc, setDoc, updateDoc, deleteDoc, collection, onSnapshot, addDoc } from 'firebase/firestore';

// রিয়েল-টাইম লিসেনার বা সাবস্ক্রিপশন ফাংশনসমূহ[cite: 12]
export const subscribeBooths = (callback) => {
  return onSnapshot(collection(db, "booths"), (snapshot) => {
    const data = snapshot.docs.map(doc => ({ id: doc.id, ...doc.data() }));
    callback(data);
  });
};

// স্টাফ ম্যানেজমেন্ট সার্ভিস[cite: 12]
export const createStaffService = async ({ newName, newEmail, newPassword, newBuses, newBranch }) => {
  let secondaryApp = null;
  try {
    secondaryApp = initializeApp(auth.app.options, "SecondaryAppForStaffCreation");
    const secondaryAuth = getAuth(secondaryApp);
    
    await createAuthUser(secondaryAuth, newEmail, newPassword);
    
    // সেফটি চেক: newBuses যদি undefined বা string না হয়, তবে অ্যাপ ক্র্যাশ করবে না[cite: 13]
    const busesArray = typeof newBuses === 'string' 
      ? newBuses.split(',').map(b => b.trim()).filter(Boolean) 
      : [];

    await setDoc(doc(db, "staff", newEmail), {
      name: newName,
      email: newEmail,
      branch: newBranch,
      buses: busesArray,
      role: 'STAFF',
      canExpense: true,
      basicSalary: 15000,
      advanceSalary: 0,
      isDeleted: false
    }, { merge: true });

    if (secondaryApp) {
      await deleteApp(secondaryApp);
    }
    return { success: true };
  } catch (error) {
    if (secondaryApp) {
      try { await deleteApp(secondaryApp); } catch (err) {}
    }
    return { success: false, error: error.message };
  }
};

export const updateStaffService = async (email, updateData) => {
  await updateDoc(doc(db, "staff", email), updateData);
};

export const setStaffDeletedStatus = async (email, isDeleted) => {
  await setDoc(doc(db, "staff", email), { isDeleted }, { merge: true });
};

export const permanentDeleteStaffService = async (email) => {
  await deleteDoc(doc(db, "staff", email));
};
