      d = new Date(dateVal);

            <h3 style={{ fontSize: '20px', marginTop: '12px', color: '#fef08a' }}>
              💰 সর্বমোট অবশিষ্ট নীট জমা: {netIncome.toLocaleString('bn-BD')} ৳
            </h3>
          </div>
        </div>
      )}

      {activeTab === 'salary' && (
        <div>
          <div style={{ display: 'flex', alignItems: 'center', justifyContent: 'space-between', flexWrap: 'wrap', gap: '15px', background: '#1e293b', padding: '16px 20px', borderRadius: '12px', border: '1px solid #334155', marginBottom: '20px' }}>
            <div style={{ display: 'flex', alignItems: 'center', gap: '10px' }}>
              <label htmlFor="monthSelect" style={{ fontWeight: '700', color: '#60a5fa', fontSize: '14px' }}>বর্তমান কার্যরত মাস নির্বাচন:</label>
              <select 
                id="monthSelect" 
                value={selectedMonth}
                onChange={(e) => {
                  setSelectedMonth(e.target.value);
                  setSalaryForm(prev => ({ ...prev, month: e.target.value }));
                }}
                style={{ padding: '8px 15px', borderRadius: '8px', border: '1px solid #334155', backgroundColor: '#0f172a', color: '#ffffff', fontWeight: 'bold', fontSize: '14px', cursor: 'pointer', outline: 'none' }}
              >
                {generateMonthOptions().map(m => (
                  <option key={m} value={m} style={{ backgroundColor: '#0f172a', color: '#ffffff' }}>{m}</option>
                ))}
              </select>
            </div>
            
            <button 
              type="button" 
              onClick={handleAutoLoadPreviousMonthSalaries}
              style={{ padding: '9px 16px', backgroundColor: '#2563eb', color: '#fff', border: 'none', borderRadius: '8px', fontWeight: '600', cursor: 'pointer', fontSize: '13px' }}>
              🔄 পূর্ববর্তী মাস থেকে স্টাফ ডাটা কপি করুন
            </button>
          </div>

          <div style={{ backgroundColor: '#1e293b', padding: '20px', borderRadius: '12px', border: '1px solid #334155', marginBottom: '25px' }}>
            <h3 style={{ margin: '0 0 14px', fontSize: '16px', fontWeight: '700', color: '#f8fafc' }}>
              ➕ স্টাফের নতুন বেতন এন্ট্রি ({selectedMonth})
            </h3>
            
            <div style={{ marginBottom: '15px', backgroundColor: '#064e3b', padding: '12px', borderRadius: '8px', border: '1px solid #065f46' }}>
              <label style={{ fontSize: '12px', fontWeight: '700', color: '#4ade80', display: 'block', marginBottom: '4px' }}>💡 শর্টকাট: নিবন্ধিত স্টাফ লিস্ট থেকে সিলেক্ট করুন</label>
              <select 
                onChange={(e) => {
                  const email = e.target.value;
                  if (!email) return;
                  const staff = staffDataMap?.[email];
                  if (staff) {
                    const dim = Number(salaryForm.daysInMonth) || 30;
                    const ad = Number(salaryForm.absentDays) || 0;
                    const b = Number(staff.basicsalary ?? staff.basicSalary) || 15000;
                    const adv = Number(staff.advancesalary ?? staff.advanceSalary) || 0;
                    const { deduction, total } = calculateSalaryDetails(b, dim, ad, adv);

                    setSalaryForm(prev => ({
                      ...prev,
                      staffName: staff.name || '',
                      branch: countersList.includes(staff.branch) ? staff.branch : countersList[0],
                      staffId: email.split('@')[0].toUpperCase(),
                      basicSalary: b,
                      advanceSalary: adv,
                      unpaidLeaveDeduction: deduction,
                      totalPayable: total
                    }));
                  }
                }}
                style={{ width: '100%', padding: '10px', borderRadius: '8px', border: '1px solid #065f46', backgroundColor: '#0f172a', fontWeight: '600', color: '#4ade80', cursor: 'pointer' }}
              >
                <option value="" style={{ backgroundColor: '#0f172a', color: '#ffffff' }}>-- স্টাফ সিলেক্ট করুন (অটো-ফিল) --</option>
                {activeStaffs.map(email => (
                  <option key={email} value={email} style={{ backgroundColor: '#0f172a', color: '#ffffff' }}>
                    {staffDataMap[email]?.name || email} — কাউন্টার: [{staffDataMap[email]?.branch || 'নেই'}]
                  </option>
                ))}
              </select>
            </div>

            <form onSubmit={handleSaveSalary} style={{ display: 'grid', gridTemplateColumns: 'repeat(auto-fit, minmax(200px, 1fr))', gap: '12px', alignItems: 'flex-end' }}>
              <div>
                <label style={{ fontSize: '12px', fontWeight: '600', color: '#cbd5e1', display: 'block', marginBottom: '4px' }}>মাস</label>
                <input type="text" name="month" value={salaryForm.month} readOnly style={{ width: '100%', padding: '10px', borderRadius: '8px', border: '1px solid #334155', backgroundColor: '#064e3b', color: '#fff' }} required />
              </div>
              <div>
                <label style={{ fontSize: '12px', fontWeight: '600', color: '#cbd5e1', display: 'block', marginBottom: '4px' }}>নাম</label>
                <input type="text" name="staffName" placeholder="যেমন: রহিম" value={salaryForm.staffName} onChange={handleSalaryInputChange} style={{ width: '100%', padding: '10px', borderRadius: '8px', border: '1px solid #334155', backgroundColor: '#0f172a', color: '#ffffff' }} required />
              </div>
              <div>
                <label style={{ fontSize: '12px', fontWeight: '600', color: '#cbd5e1', display: 'block', marginBottom: '4px' }}>মূল কাউন্টার</label>
                <select name="branch" value={salaryForm.branch} onChange={handleSalaryInputChange} style={{ width: '100%', padding: '10px', borderRadius: '8px', border: '1px solid #334155', backgroundColor: '#0f172a', color: '#ffffff', fontWeight: 'bold' }} required>
                  {countersList.map(c => <option key={c} value={c} style={{ backgroundColor: '#0f172a', color: '#ffffff' }}>{c}</option>)}
                </select>
              </div>
              <div>
                <label style={{ fontSize: '12px', fontWeight: '600', color: '#cbd5e1', display: 'block', marginBottom: '4px' }}>স্টাফ আইডি</label>
                <input type="text" name="staffId" placeholder="AKT 26-001" value={salaryForm.staffId} onChange={handleSalaryInputChange} style={{ width: '100%', padding: '10px', borderRadius: '8px', border: '1px solid #334155', backgroundColor: '#0f172a', color: '#ffffff' }} required />
              </div>
              <div>
                <label style={{ fontSize: '12px', fontWeight: '600', color: '#cbd5e1', display: 'block', marginBottom: '4px' }}>পদবী</label>
                <input type="text" name="designation" placeholder="ম্যানেজার" value={salaryForm.designation} onChange={handleSalaryInputChange} style={{ width: '100%', padding: '10px', borderRadius: '8px', border: '1px solid #334155', backgroundColor: '#0f172a', color: '#ffffff' }} required />
              </div>
              <div>
                <label style={{ fontSize: '12px', fontWeight: '600', color: '#cbd5e1', display: 'block', marginBottom: '4px' }}>মূল বেতন (৳)</label>
                <input type="number" name="basicSalary" placeholder="১৫০০০" value={salaryForm.basicSalary} onChange={handleSalaryInputChange} onWheel={(e) => e.target.blur()} style={{ width: '100%', padding: '10px', borderRadius: '8px', border: '1px solid #334155', backgroundColor: '#0f172a', color: '#ffffff' }} required />
              </div>
              <div>
                <label style={{ fontSize: '12px', fontWeight: '600', color: '#cbd5e1', display: 'block', marginBottom: '4px' }}>অনুপস্থিত দিন</label>
                <input type="number" name="absentDays" value={salaryForm.absentDays} onChange={handleSalaryInputChange} onWheel={(e) => e.target.blur()} style={{ width: '100%', padding: '10px', borderRadius: '8px', border: '1px solid #334155', backgroundColor: '#0f172a', color: '#ffffff' }} />
              </div>
              <div>
                <label style={{ fontSize: '12px', fontWeight: '600', color: '#cbd5e1', display: 'block', marginBottom: '4px' }}>অগ্রিম বেতন কর্তন (৳)</label>
                <input type="number" name="advanceSalary" value={salaryForm.advanceSalary} onChange={handleSalaryInputChange} onWheel={(e) => e.target.blur()} style={{ width: '100%', padding: '10px', borderRadius: '8px', border: '1px solid #334155', backgroundColor: '#0f172a', color: '#ffffff' }} />
              </div>
              <div>
                <label style={{ fontSize: '12px', fontWeight: '600', color: '#cbd5e1', display: 'block', marginBottom: '4px' }}>সর্বমোট প্রদেয় বেতন (৳)</label>
                <input type="number" name="totalPayable" value={salaryForm.totalPayable} onChange={handleSalaryInputChange} onWheel={(e) => e.target.blur()} style={{ width: '100%', padding: '10px', borderRadius: '8px', border: '1px solid #334155', backgroundColor: '#064e3b', color: '#4ade80', fontWeight: 'bold' }} required />
              </div>
              <div>
                <label style={{ fontSize: '12px', fontWeight: '600', color: '#cbd5e1', display: 'block', marginBottom: '4px' }}>পরিশোধ স্ট্যাটাস</label>
                <select name="paymentStatus" value={salaryForm.paymentStatus} onChange={handleSalaryInputChange} style={{ width: '100%', padding: '10px', borderRadius: '8px', border: '1px solid #334155', backgroundColor: '#0f172a', color: '#ffffff' }}>
                  <option value="অপরিশোধ" style={{ backgroundColor: '#0f172a', color: '#ffffff' }}>অপরিশোধ</option>
                  <option value="পরিশোধিত" style={{ backgroundColor: '#0f172a', color: '#ffffff' }}>পরিশোধিত</option>
                </select>
              </div>
              <button type="submit" disabled={isSavingSalary} style={{ gridColumn: '1 / -1', padding: '12px', backgroundColor: isSavingSalary ? '#64748b' : '#16a34a', color: '#fff', border: 'none', borderRadius: '8px', fontWeight: '600', cursor: isSavingSalary ? 'not-allowed' : 'pointer' }}>
                {isSavingSalary ? 'জমা হচ্ছে...' : 'বেতনের রেকর্ড সেভ করুন'}
              </button>
            </form>
          </div>

          {/* স্টাফ বেতন রেকর্ডসমূহ ড্রপডাউন / টগল সেকশন */}
          <div style={{ backgroundColor: '#1e293b', borderRadius: '12px', border: '1px solid #334155', overflow: 'hidden', marginBottom: '20px' }}>
            <div 
              onClick={() => setIsSalaryTableOpen(prev => !prev)}
              style={{ padding: '14px 20px', display: 'flex', justifyContent: 'space-between', alignItems: 'center', cursor: 'pointer', backgroundColor: '#0f172a', userSelect: 'none' }}
            >
              <h3 style={{ margin: 0, fontSize: '15px', fontWeight: '700', color: '#f8fafc', display: 'flex', alignItems: 'center', gap: '8px' }}>
                <span>{isSalaryTableOpen ? '▼' : '▶'}</span>
                📋 স্টাফ বেতন রেকর্ডসমূহ (কাউন্টার ওয়াইস ভাগ করা) ({selectedMonth})
              </h3>
              <button 
                type="button" 
                style={{ fontSize: '12px', color: '#ffffff', fontWeight: '600', backgroundColor: '#334155', border: '1px solid #475569', padding: '6px 12px', borderRadius: '6px', cursor: 'pointer' }}
              >
                {isSalaryTableOpen ? 'ড্রপডাউন বন্ধ করুন ▴' : 'ড্রপডাউন খুলুন ▾'}
              </button>
            </div>

            {isSalaryTableOpen && (
              <div style={{ padding: '16px 20px' }}>
                {activeSalaries.filter(s => s.month === selectedMonth).length === 0 ? (
                  <p style={{ fontSize: '13px', color: '#94a3b8', fontStyle: 'italic', margin: 0 }}>এই মাসে কোনো বেতনের রেকর্ড নেই।</p>
                ) : (
                  ALL_REPORT_COUNTERS.map(counterName => {
                    const counterSalaries = activeSalaries.filter(s => s.month === selectedMonth && (s.branch === counterName || (!countersList.includes(s.branch) && counterName === "অনির্দিষ্ট/অন্যান্য কাউন্টার")));
                    if (counterSalaries.length === 0) return null;

                    return (
                      <div key={counterName} style={{ marginBottom: '20px', backgroundColor: '#1e293b', borderRadius: '12px', border: '1px solid #334155', overflow: 'hidden' }}>
                        <div style={{ backgroundColor: '#0f172a', padding: '12px 18px', fontWeight: 'bold', color: '#60a5fa', borderBottom: '1px solid #334155', display: 'flex', justifyContent: 'space-between', alignItems: 'center' }}>
                          <span>📍 কাউন্টার: {counterName}</span>
                          <span style={{ fontSize: '12px', backgroundColor: '#334155', padding: '4px 8px', borderRadius: '6px', color: '#f8fafc' }}>
                            মোট স্টাফ: {counterSalaries.length} জন | মোট প্রদেয়: ৳{counterSalaries.reduce((sum, s) => sum + (Number(s.totalPayable) || 0), 0).toLocaleString('bn-BD')}
                          </span>
                        </div>
                        <div style={{ padding: '15px', overflowX: 'auto' }}>
                          <table style={{ width: '100%', borderCollapse: 'collapse', textAlign: 'left', fontSize: '13px', color: '#f8fafc' }}>
                            <thead>
                              <tr style={{ backgroundColor: '#334155', color: '#f8fafc' }}>
                                <th style={{ padding: '10px', border: '1px solid #475569' }}>নাম</th>
                                <th style={{ padding: '10px', border: '1px solid #475569' }}>আইডি</th>
                                <th style={{ padding: '10px', border: '1px solid #475569' }}>ব্রাঞ্চ</th>
                                <th style={{ padding: '10px', border: '1px solid #475569' }}>পদবী</th>
                                <th style={{ padding: '10px', border: '1px solid #475569' }}>প্রদেয় বেতন (৳)</th>
                                <th style={{ padding: '10px', border: '1px solid #475569' }}>স্ট্যাটাস</th>
                                <th style={{ padding: '10px', border: '1px solid #475569', textAlign: 'center' }}>অ্যাকশন</th>
                              </tr>
                            </thead>
                            <tbody>
                              {counterSalaries.map(sal => (
                                <tr key={sal.id} style={{ borderBottom: '1px solid #334155' }}>
                                  <td style={{ padding: '10px', border: '1px solid #475569' }}>{sal.staffName}</td>
                                  <td style={{ padding: '10px', border: '1px solid #475569' }}>{sal.staffId}</td>
                                  <td style={{ padding: '10px', border: '1px solid #475569' }}>{sal.branch}</td>
                                  <td style={{ padding: '10px', border: '1px solid #475569' }}>{sal.designation}</td>
                                  <td style={{ padding: '10px', border: '1px solid #475569', fontWeight: 'bold', color: '#4ade80' }}>৳{sal.totalPayable}</td>
                                  <td style={{ padding: '10px', border: '1px solid #475569' }}>
                                    <span style={{ padding: '4px 8px', borderRadius: '4px', fontSize: '11px', backgroundColor: sal.paymentStatus === 'পরিশোধিত' ? '#064e3b' : '#7c2d12', color: '#fff' }}>
                                      {sal.paymentStatus || 'অপরিশোধ'}
                                    </span>
                                  </td>
                                  <td style={{ padding: '10px', border: '1px solid #475569', textAlign: 'center' }}>
                                    <button onClick={() => handleDeleteSalary(sal.id)} style={{ padding: '5px 10px', backgroundColor: '#dc2626', color: '#fff', border: 'none', borderRadius: '6px', cursor: 'pointer' }}>ডিলিট</button>
                                  </td>
                                </tr>
                              ))}
                            </tbody>
                          </table>
                        </div>
                      </div>
                    );
                  })
                )}
              </div>
            )}
          </div>
        </div>
      )}

    </div>
  );
}
